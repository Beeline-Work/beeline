// FILE: database.kt
package expo.modules.updates.db
import expo.modules.updates.db.entity.*
import expo.modules.updates.db.enums.UpdateStatus
import java.util.UUID
class UpdatesDatabase(val updates: MutableList<UpdateEntity>, val assets: MutableMap<UUID, AssetEntity>) {
  fun updateDao() = UpdateDao(this)
  fun assetDao() = AssetDao(this)
}
class UpdateDao(val db: UpdatesDatabase) {
  fun loadLaunchableUpdatesForScope(scope: String) = db.updates.filter { it.status != UpdateStatus.PENDING }
  fun loadLaunchAssetForUpdate(id: UUID) = db.assets[id]
  fun markUpdateAccessed(update: UpdateEntity) {}
  fun markUpdatesWithMissingAssets(assets: List<AssetEntity>) {
    db.updates.filter { db.assets[it.id] in assets }.forEach { it.status = UpdateStatus.PENDING }
  }
  fun loadUpdateWithId(id: UUID) = db.updates.find { it.id == id }
  fun insertUpdate(update: UpdateEntity) { db.updates.add(update) }
  fun markUpdateFinished(update: UpdateEntity) { update.status = UpdateStatus.READY }
  fun setUpdateScopeKey(update: UpdateEntity, scope: String) { update.scopeKey = scope }
}
class AssetDao(val db: UpdatesDatabase) {
  fun loadAssetsForUpdate(id: UUID) = listOfNotNull(db.assets[id])
  fun updateAsset(asset: AssetEntity) {}
  fun loadAssetWithKey(key: String?) = db.assets.values.find { it.key == key }
  fun mergeAndUpdateAsset(existing: AssetEntity, asset: AssetEntity) {}
  fun addExistingAssetToUpdate(update: UpdateEntity, asset: AssetEntity, isLaunch: Boolean) = true
  fun insertAssets(assets: List<AssetEntity>, update: UpdateEntity) { assets.firstOrNull()?.let { db.assets[update.id] = it } }
}
